# Mac App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> Tim's own rule overrides the line above: build this plan with `pstack:poteto-mode`.

**Goal:** Package iconmaschine as `iconmaschine.app` (Electron) that starts its own Next.js server, per `docs/prd-mac-app.md`.

**Architecture:** `next build` with `output: "standalone"` makes a self-contained server folder. electron-builder copies that folder into the app's `Resources/server`. `electron/main.cjs` starts it with `utilityProcess.fork` on a free `127.0.0.1` port, waits for `/api/status`, then opens a window on it. API keys move from `process.env` only to `src/lib/settings.ts`, which reads the environment first and `~/.iconmaschine/settings.json` second.

**Tech Stack:** Next.js 16.4.5, Electron 45.2.3 (Node 24.22.2), electron-builder 26.18.0, sharp 0.35.5, `node:sqlite`.

**Facts proven on 2026-11-08 (scratch build):** `node:sqlite` and `sharp` work under Electron 45. The standalone server (52 MB) runs under Electron's Node and serves `/`, `/api/status`, `/api/packs` and `/api/tagvocab`. The standalone folder holds `@img/sharp-darwin-arm64` and `@img/sharp-libvips-darwin-arm64`, and no `.env*` file.

```ts
// src/lib/settings.ts
import fs from "node:fs";
import fsp from "node:fs/promises";
import path from "node:path";
import { iconmaschineHome } from "./paths";

export const SETTING_NAMES = ["OPENAI_API_KEY", "SERPAPI_API_KEY", "OPENAI_IMAGE_MODEL"] as const;
export type SettingName = (typeof SETTING_NAMES)[number];
export type Settings = Partial<Record<SettingName, string>>;

export interface SettingStatus {
  set: boolean;
  source: "env" | "file" | null;
  /** The last four characters, so Tim can tell two keys apart. */
  hint: string | null;
}

const file = () => path.join(iconmaschineHome(), "settings.json");

/** Read on every call, so a saved key works at once, without a restart. */
function readFile(): Settings {
  try {
    const raw = JSON.parse(fs.readFileSync(file(), "utf8")) as Record<string, unknown>;
    const out: Settings = {};
    for (const n of SETTING_NAMES) {
      const v = raw[n];
      if (typeof v === "string" && v.trim()) out[n] = v.trim();
    }
    return out;
  } catch {
    return {};
  }
}

/** The environment wins, so `.env.local` keeps working in dev. */
export function setting(name: SettingName): string | undefined {
  const env = process.env[name]?.trim();
  return env || readFile()[name];
}

export function settingsStatus(): Record<SettingName, SettingStatus> {
  const stored = readFile();
  const out = {} as Record<SettingName, SettingStatus>;
  for (const n of SETTING_NAMES) {
    const env = process.env[n]?.trim();
    const value = env || stored[n];
    out[n] = {
      set: Boolean(value),
      source: env ? "env" : stored[n] ? "file" : null,
      hint: value ? value.slice(-4) : null,
    };
  }
  return out;
}

/** Merges into the saved file. An empty string clears that one setting. */
export async function saveSettings(changes: Settings): Promise<void> {
  const next = readFile();
  for (const n of SETTING_NAMES) {
    if (!(n in changes)) continue;
    const v = changes[n]?.trim();
    if (v) next[n] = v;
    else delete next[n];
  }
  await fsp.mkdir(iconmaschineHome(), { recursive: true });
  const tmp = `${file()}.tmp`;
  await fsp.writeFile(tmp, JSON.stringify(next, null, 2), { mode: 0o600 });
  await fsp.chmod(tmp, 0o600);
  await fsp.rename(tmp, file());
}
```

```ts
export async function POST(req: Request) {
  const body = (await req.json().catch(() => null)) as Record<string, unknown> | null;
  if (!body || typeof body !== "object") {
    return Response.json({ error: "Send a JSON object." }, { status: 400 });

  const changes: Settings = {};
  for (const n of SETTING_NAMES) {
    if (typeof body[n] === "string") changes[n] = body[n] as string;

  return Response.json({ settings: settingsStatus(), home: iconmaschineHome() });
}
```

```tsx
// src/components/SettingsPanel.tsx
"use client";

import { useEffect, useState } from "react";
import { getJson, postJson } from "@/lib/client";

type Name = "OPENAI_API_KEY" | "SERPAPI_API_KEY" | "OPENAI_IMAGE_MODEL";
interface Status { set: boolean; source: "env" | "file" | null; hint: string | null }
interface Res { settings: Record<Name, Status>; home: string }

const FIELDS: { name: Name; label: string; help: string; secret: boolean }[] = [
  { name: "OPENAI_API_KEY", label: "OpenAI API key", help: "Used only by Generate with AI.", secret: true },
  { name: "SERPAPI_API_KEY", label: "SerpAPI key", help: "Used only by Search for a logo. serpapi.com has a free tier.", secret: true },
  { name: "OPENAI_IMAGE_MODEL", label: "Image model (optional)", help: "Leave empty to use the newest model your account has.", secret: false },
];

/** Keys are saved on this Mac only, in settings.json in the app's data folder. */
export function SettingsPanel({ onSaved }: { onSaved: () => void }) {
  const [res, setRes] = useState<Res | null>(null);
  const [draft, setDraft] = useState<Partial<Record<Name, string>>>({});
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [savedNote, setSavedNote] = useState(false);

  useEffect(() => {
    getJson<Res>("/api/settings").then(setRes, (e) => setError((e as Error).message));
  }, []);

  async function save(changes: Partial<Record<Name, string>>) {
    setBusy(true);
    setError(null);
    setSavedNote(false);
    try {
      setRes(await postJson<Res>("/api/settings", changes));
      setDraft({});
      setSavedNote(true);
      onSaved();
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  const dirty = Object.values(draft).some((v) => v !== undefined && v !== "");

  return (
    <div>
      <p className="hint">
        Keys are saved on this Mac only, in <code>{res?.home ?? "~/.iconmaschine"}/settings.json</code>.
        Only your user account can read that file.
      </p>
      {FIELDS.map((f) => {
        const s = res?.settings[f.name];
        return (
          <div className="field" key={f.name}>
            <label htmlFor={f.name}>{f.label}</label>
            <input
              id={f.name}
              type={f.secret ? "password" : "text"}
              autoComplete="off"
              placeholder={s?.set ? `Set, ends in …${s.hint}` : "Not set"}
              value={draft[f.name] ?? ""}
              onChange={(e) => setDraft((d) => ({ ...d, [f.name]: e.target.value }))}
            />
            <div className="hint">
              {f.help}
              {s?.source === "env" && " Set by an environment variable, which wins over this field."}
              {s?.source === "file" && (
                <>
                  {" "}
                  <button type="button" className="link-btn" disabled={busy} onClick={() => save({ [f.name]: "" })}>
                    Remove
                  </button>
                </>
              )}
            </div>
          </div>
        );
      })}
      <div className="actions">
        <button className="btn" disabled={busy || !dirty} onClick={() => save(draft)}>
          {busy ? <><span className="spinner" /> Saving…</> : "Save"}
        </button>
      </div>
      {savedNote && <div className="note">Saved. The new values work now, with no restart.</div>}
      {error && <div className="note bad">{error}</div>}
    </div>
  );
}
```

```js
// scripts/prepare-app.mjs
// Runs after `next build` with ICONMASCHINE_STANDALONE=1. Makes the
// standalone folder complete, makes the app icon, and refuses a build that
// would ship keys or miss a native part.
import fs from "node:fs";
import path from "node:path";
import sharp from "sharp";

const root = process.cwd();
const sa = path.join(root, ".next", "standalone");
const fail = (msg) => {
  console.error(`prepare-app: ${msg}`);
  process.exit(1);
};

if (!fs.existsSync(path.join(sa, "server.js"))) {
  fail(".next/standalone/server.js is missing. Did next build run with ICONMASCHINE_STANDALONE=1?");
}

// The standalone server serves these only if they sit inside it.
fs.cpSync(path.join(root, ".next", "static"), path.join(sa, ".next", "static"), { recursive: true });
if (fs.existsSync(path.join(root, "public"))) {
  fs.cpSync(path.join(root, "public"), path.join(sa, "public"), { recursive: true });
}

// Never ship keys or local data.
const banned = [];
const walk = (dir) => {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) {
      if (e.name !== "node_modules") walk(p);
    } else if (/^\.env/.test(e.name) || e.name === "settings.json") {
      banned.push(path.relative(root, p));
    }
  }
};
walk(sa);
if (banned.length) fail(`refusing to package secrets or local data: ${banned.join(", ")}`);

// sharp needs its native parts for this chip.
for (const pkg of ["@img/sharp-darwin-arm64", "@img/sharp-libvips-darwin-arm64"]) {
  if (!fs.existsSync(path.join(sa, "node_modules", pkg))) fail(`${pkg} is missing from the standalone build.`);
}

const res = path.join(root, "electron", "resources");
fs.mkdirSync(res, { recursive: true });
await sharp(path.join(root, "electron", "icon.svg")).resize(1024, 1024).png().toFile(path.join(res, "icon.png"));

console.log("prepare-app: standalone folder ready, icon made.");
```

```js
// electron/main.cjs
// Starts the Next.js standalone server on a free 127.0.0.1 port, then shows
// it in a window. Quitting the app stops the server.
const { app, BrowserWindow, Menu, shell, utilityProcess } = require("electron");
const fs = require("node:fs");
const http = require("node:http");
const net = require("node:net");
const path = require("node:path");

const SMOKE = process.env.ICONMASCHINE_SMOKE === "1";
let win = null;
let server = null;
let origin = null;
let quitting = false;

if (!app.requestSingleInstanceLock()) {
  app.quit();
} else {
  app.on("second-instance", () => {
    if (!win) return;
    if (win.isMinimized()) win.restore();
    win.focus();
  });
  app.whenReady().then(main);
}

function freePort() {
  return new Promise((resolve, reject) => {
    const s = net.createServer();
    s.unref();
    s.on("error", reject);
    s.listen(0, "127.0.0.1", () => {
      const { port } = s.address();
      s.close(() => resolve(port));
    });
  });
}

function ping(url) {
  return new Promise((resolve) => {
    const req = http.get(url, (res) => {
      res.resume();
      resolve(res.statusCode < 500);
    });
    req.on("error", () => resolve(false));
    req.setTimeout(1000, () => req.destroy());
  });
}

async function waitFor(url, ms) {
  const end = Date.now() + ms;
  while (Date.now() < end) {
    if (await ping(url)) return true;
    await new Promise((r) => setTimeout(r, 200));
  }
  return false;
}

function serverDir() {
  return app.isPackaged
    ? path.join(process.resourcesPath, "server")
    : path.join(__dirname, "..", ".next", "standalone");
}

function logFile() {
  const dir = app.getPath("logs");
  fs.mkdirSync(dir, { recursive: true });
  return path.join(dir, "server.log");
}

async function startServer(log) {
  const port = await freePort();
  const dir = serverDir();
  const out = fs.createWriteStream(log, { flags: "a" });
  out.write(`\n--- ${new Date().toISOString()} start on port ${port}\n`);
  server = utilityProcess.fork(path.join(dir, "server.js"), [], {
    cwd: dir,
    stdio: "pipe",
    serviceName: "iconmaschine server",
    env: {
      ...process.env,
      PORT: String(port),
      HOSTNAME: "127.0.0.1",
      NODE_ENV: "production",
      NEXT_TELEMETRY_DISABLED: "1",
    },
  });
  server.stdout?.pipe(out);
  server.stderr?.pipe(out);
  server.on("exit", (code) => {
    server = null;
    if (!quitting) showError(`The server stopped (exit code ${code}).`, log);
  });
  origin = `http://127.0.0.1:${port}`;
  return waitFor(`${origin}/api/status`, 30000);
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]);
}

function showError(message, log) {
  if (SMOKE) {
    console.error(`smoke: ${message}`);
    app.exit(1);
    return;
  }
  if (!win) createWindow();
  const html = `<!doctype html><meta charset="utf-8"><body style="font:14px -apple-system;background:#141414;color:#eee;padding:40px">
<h2>iconmaschine could not start</h2><p>${escapeHtml(message)}</p>
<p>The server log is at <code>${escapeHtml(log)}</code>.</p></body>`;
  win.loadURL(`data:text/html;charset=utf-8,${encodeURIComponent(html)}`);
}

function isLocal(url) {
  return origin !== null && url.startsWith(origin + "/");
}

function createWindow() {
  win = new BrowserWindow({
    width: 1320,
    height: 900,
    minWidth: 380,
    title: "iconmaschine",
    backgroundColor: "#141414",
    webPreferences: { contextIsolation: true, sandbox: true },
  });
  // Links to other sites open in Tim's normal browser.
  win.webContents.setWindowOpenHandler(({ url }) => {
    if (!isLocal(url)) shell.openExternal(url);
    return { action: "deny" };
  });
  win.webContents.on("will-navigate", (event, url) => {
    if (!isLocal(url) && !url.startsWith("data:")) {
      event.preventDefault();
      shell.openExternal(url);
    }
  });
  win.on("closed", () => {
    win = null;
  });
}

async function smokeCheck() {
  const pages = ["/", "/api/status", "/api/packs"];
  for (const p of pages) {
    const ok = await ping(origin + p);
    console.log(`smoke: ${p} ${ok ? "ok" : "FAILED"}`);
    if (!ok) return app.exit(1);
  }
  quitting = true;
  server?.kill();
  app.exit(0);
}

async function main() {
  Menu.setApplicationMenu(
    Menu.buildFromTemplate([
      { role: "appMenu" },
      { role: "editMenu" },
      { role: "viewMenu" },
      { role: "windowMenu" },
    ]),
  );
  const log = logFile();
  const ready = await startServer(log).catch((e) => {
    showError(e.message, log);
    return null;
  });
  if (ready === null) return;
  if (!ready) return showError("The server did not answer within 30 seconds.", log);
  if (SMOKE) return smokeCheck();
  if (!win) createWindow();
  win.loadURL(origin + "/");
}

app.on("before-quit", () => {
  quitting = true;
  server?.kill();
});

// The server runs only while the window is open (PRD non-goal: no background server).
app.on("window-all-closed", () => app.quit());
```
